# Sqorz Hub — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Créer un `index.html` autonome qui sert de hub GitHub Pages listant automatiquement tous les outils Sqorz du compte `ludsoc`, avec le même design system que les projets existants.

**Architecture:** Single fichier `index.html` — CSS et JS entièrement embarqués, aucune dépendance externe, aucun build. Au chargement, deux types d'appels API sont effectués côté client : (1) GitHub API pour lister les repos avec `has_pages: true`, (2) `raw.githubusercontent.com` pour extraire le SVG logo de chaque projet.

**Tech Stack:** HTML5, CSS (variables, flexbox, keyframes), JavaScript (fetch, DOMParser, Intl, localStorage), GitHub REST API v3 (unauthenticated, public repos).

> **Note opérationnelle :** Ne pas créer de repo GitHub ni committer. Écrire uniquement `index.html` localement. L'utilisateur gère lui-même git.

```html
<script>
  const THEME_ICONS = { auto: '🌓', light: '☀️', dark: '🌙' };
  const themeBtn = document.getElementById('themeBtn');

  function currentPref() {
    return localStorage.getItem('theme') || 'auto';
  }

  function applyTheme(pref) {
    const isDark = pref === 'dark' ||
      (pref === 'auto' && window.matchMedia('(prefers-color-scheme: dark)').matches);
    document.documentElement.classList.toggle('dark', isDark);
    if (pref === 'auto') delete document.documentElement.dataset.themePref;
    else document.documentElement.dataset.themePref = pref;
    themeBtn.textContent = THEME_ICONS[pref];
    if (pref === 'auto') localStorage.removeItem('theme');
    else localStorage.setItem('theme', pref);
  }

  function nextPref(p) {
    return p === 'auto' ? 'light' : p === 'light' ? 'dark' : 'auto';
  }

  themeBtn.addEventListener('click', () => applyTheme(nextPref(currentPref())));
  applyTheme(currentPref());
</script>
```

```javascript
const GH_USER = 'ludsoc';
const HUB_REPO = 'sqorz-hub';
const ICON_GRADIENTS = [
  'linear-gradient(135deg,#2e4158,#6a8caf)',
  'linear-gradient(135deg,#3f566f,#8aa6c2)',
  'linear-gradient(135deg,#2e4158,#88ccd0)',
  'linear-gradient(135deg,#506f8f,#9bb5cf)',
];

async function fetchRepos() {
  const res = await fetch(
    `https://api.github.com/users/${GH_USER}/repos?per_page=100&sort=updated`
  );
  if (!res.ok) throw new Error(`GitHub API ${res.status}`);
  const repos = await res.json();
  return repos
    .filter(r => r.has_pages && r.name !== HUB_REPO)
    .sort((a, b) => new Date(b.updated_at) - new Date(a.updated_at));
}
```

```javascript
async function fetchIcon(repoName) {
  try {
    const res = await fetch(
      `https://raw.githubusercontent.com/${GH_USER}/${repoName}/main/index.html`
    );
    if (!res.ok) return null;
    const html = await res.text();
    const doc = new DOMParser().parseFromString(html, 'text/html');
    const logo = doc.querySelector('.logo');
    const svg = logo?.querySelector('svg');
    return svg ? svg.outerHTML : null;
  } catch {
    return null;
  }
}

function initials(name) {
  return name.split(/[-_]/)
    .map(w => w[0]?.toUpperCase() ?? '')
    .join('')
    .slice(0, 2) || name.slice(0, 2).toUpperCase();
}
```

```javascript
function formatDate(iso) {
  return new Intl.DateTimeFormat('fr-FR', {
    day: 'numeric', month: 'short', year: 'numeric'
  }).format(new Date(iso));
}

function renderCard(repo, icon, gradient) {
  const iconHtml = icon
    ? icon
    : `<span style="font-size:15px;font-weight:800">${initials(repo.name)}</span>`;
  const desc = repo.description || 'Aucune description disponible';
  const url = `https://${GH_USER}.github.io/${repo.name}/`;
  return `
    <div class="project-card">
      <div class="project-icon" style="background:${gradient}" aria-hidden="true">
        ${iconHtml}
      </div>
      <div class="project-info">
        <div class="project-name">${repo.name}</div>
        <div class="project-desc" title="${desc}">${desc}</div>
        <div class="project-meta">
          <svg width="11" height="11" viewBox="0 0 16 16" fill="currentColor">
            <path d="M8 0a8 8 0 1 1 0 16A8 8 0 0 1 8 0zm.5 4.75a.75.75 0 0 0-1.5 0v3.5a.75.75 0 0 0 .22.53l2.25 2.25a.75.75 0 1 0 1.06-1.06L8.5 7.94V4.75z"/>
          </svg>
          Mis à jour le ${formatDate(repo.updated_at)}
        </div>
      </div>
      <a class="project-btn" href="${url}" target="_blank" rel="noopener">Ouvrir →</a>
    </div>
  `;
}

async function renderProjects() {
  renderSkeleton();
  try {
    const repos = await fetchRepos();
    const icons = await Promise.all(repos.map(r => fetchIcon(r.name)));
    const n = repos.length;
    const label = n === 0
      ? 'Aucun outil disponible'
      : `${n} outil${n > 1 ? 's' : ''} disponible${n > 1 ? 's' : ''}`;
    const cards = repos
      .map((r, i) => renderCard(r, icons[i], ICON_GRADIENTS[i % 4]))
      .join('');
    document.getElementById('main').innerHTML = `
      <div class="status-bar">
        <div class="status-dot"></div>
        ${label} · mis à jour il y a quelques secondes
        <span style="margin-left:auto;opacity:.6">via GitHub API</span>
      </div>
      <div class="project-list">${cards}</div>
      <div class="hub-footer">
        Source : <a href="https://github.com/${GH_USER}" target="_blank" rel="noopener">github.com/${GH_USER}</a>
        · GitHub Pages détectées automatiquement
      </div>
    `;
  } catch (err) {
    document.getElementById('main').innerHTML = `
      <div class="error-card">
        <p>Impossible de récupérer la liste des projets (${err.message}).</p>
        <p>Voir directement
          <a href="https://github.com/${GH_USER}" target="_blank" rel="noopener">
            github.com/${GH_USER}
          </a>.
        </p>
      </div>
    `;
  }
}
```
